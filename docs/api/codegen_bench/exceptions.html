codegen_bench.exceptions

Custom exceptions for BrownBench.

Error Taxonomy:

All errors inherit from BrownBenchError and include:

  1. Detailed error messages with context
  2. Actionable suggestions for resolution
  3. Error metadata for reporting and analysis
  4. Retry capability indication

Error Categories:

  • Configuration Errors: Invalid config, missing dependencies (not retryable)
  • Transient Errors: Network issues, rate limits (retryable)
  • Data Errors: Invalid task data, parsing failures (not retryable)
  • Infrastructure Errors: Git, Docker, filesystem issues (may be retryable)
  • API Errors: External API failures (retryable with backoff)

 80class BrownBenchError(Exception):
 81    """Base exception for all BrownBench errors.
 82
 83    All BrownBench exceptions include:
 84    - Rich context about the error
 85    - Actionable suggestions for resolution
 86    - Metadata for error reporting and analysis
 87    """
 88
 89    def __init__(
 90        self,
 91        message: str,
 92        *,
 93        category: ErrorCategory,
 94        severity: ErrorSeverity = ErrorSeverity.ERROR,
 95        is_retryable: bool = False,
 96        retry_suggestion: str | None = None,
 97        component: str | None = None,
 98        operation: str | None = None,
 99        task_id: str | None = None,
100        dut_id: str | None = None,
101        **metadata: Any,
102    ):
103        """Initialize BrownBenchError.
104
105        Args:
106            message: Human-readable error message
107            category: Error category
108            severity: Error severity level
109            is_retryable: Whether this error can be retried
110            retry_suggestion: How to retry (if retryable)
111            component: Component that raised the error
112            operation: Operation that failed
113            task_id: Task ID if applicable
114            dut_id: DUT ID if applicable
115            **metadata: Additional metadata
116        """
117        super().__init__(message)
118        self.context = ErrorContext(
119            timestamp=datetime.now(UTC).isoformat(),
120            category=category,
121            severity=severity,
122            is_retryable=is_retryable,
123            retry_suggestion=retry_suggestion,
124            component=component,
125            operation=operation,
126            task_id=task_id,
127            dut_id=dut_id,
128            metadata=metadata,
129        )
130
131    def get_actionable_message(self) -> str:
132        """Get error message with actionable suggestions.
133
134        Returns:
135            Formatted error message with resolution steps
136        """
137        msg = f"{self}\n"
138        msg += f"Category: {self.context.category.value}\n"
139        msg += f"Severity: {self.context.severity.value}\n"
140
141        if self.context.component:
142            msg += f"Component: {self.context.component}\n"
143        if self.context.operation:
144            msg += f"Operation: {self.context.operation}\n"
145
146        if self.context.is_retryable:
147            msg += "\nThis error is retryable.\n"
148            if self.context.retry_suggestion:
149                msg += f"Suggestion: {self.context.retry_suggestion}\n"
150        else:
151            msg += "\nThis error is not retryable. Manual intervention required.\n"
152
153        return msg

297class DUTAPIError(DUTError):
298    """Raised when DUT API calls fail.
299
300    Examples:
301    - Anthropic API errors
302    - HTTP 500 errors
303    - Authentication failures
304    """
305
306    def __init__(self, message: str, **kwargs: Any):
307        # Determine if retryable based on HTTP status
308        http_status = kwargs.get("http_status")
309        if "is_retryable" not in kwargs:
310            # Non-retryable status codes
311            kwargs["is_retryable"] = (
312                http_status not in (400, 401, 403, 404) if http_status else True
313            )
314
315        if "category" not in kwargs:
316            kwargs["category"] = ErrorCategory.API
317        if "component" not in kwargs:
318            kwargs["component"] = "DUT"
319        if "operation" not in kwargs:
320            kwargs["operation"] = "api_call"
321        if "retry_suggestion" not in kwargs and kwargs.get("is_retryable"):
322            kwargs["retry_suggestion"] = "Retry with exponential backoff (1s, 2s, 4s, 8s, 16s)"
323
324        super().__init__(message, **kwargs)
