def analyze_pareto_frontier( objectives: numpy.ndarray[tuple[typing.Any, ...], numpy.dtype[numpy.float64]], reference_point: numpy.ndarray[tuple[typing.Any, ...], numpy.dtype[numpy.float64]] | None = None) -> ParetoResult: 
125def analyze_pareto_frontier(
126    objectives: npt.NDArray[np.float64],
127    reference_point: npt.NDArray[np.float64] | None = None,
128) -> ParetoResult:
129    """Perform complete Pareto frontier analysis.
130
131    Args:
132        objectives: Array of objective values (n_points x n_objectives)
133            All objectives assumed to be minimized
134        reference_point: Reference point for hypervolume (worst values)
135            If None, uses max values + 10% margin
136
137    Returns:
138        ParetoResult with frontier, hypervolume, and domination info
139    """
140    # Find Pareto front
141    pareto_indices = find_pareto_front(objectives)
142    pareto_front = objectives[pareto_indices]
143
144    # Calculate reference point if not provided
145    if reference_point is None:
146        max_values = np.max(objectives, axis=0)
147        # Ensure reference point strictly dominates all points
148        # Use additive margin (10% + 0.1) to handle near-zero values
149        # This ensures even objectives near 0 get a reasonable margin
150        reference_point = max_values + np.maximum(max_values * 0.1, 0.1)
151
152    # Calculate hypervolume
153    hv = calculate_hypervolume(pareto_front, reference_point)
154
155    # Create domination mask
156    dominated_mask = np.ones(len(objectives), dtype=bool)
157    dominated_mask[pareto_indices] = False
158
159    return ParetoResult(
160        pareto_indices=pareto_indices,
161        pareto_front=pareto_front,
162        hypervolume=hv,
163        dominated_mask=dominated_mask,
164        reference_point=reference_point,
165    )

 88def calculate_hypervolume(
 89    pareto_front: npt.NDArray[np.float64],
 90    reference_point: npt.NDArray[np.float64],
 91) -> float:
 92    """Calculate hypervolume indicator for the Pareto front.
 93
 94    The hypervolume is the volume of the space dominated by the Pareto front
 95    and bounded by the reference point. Larger values indicate better coverage.
 96
 97    This implementation uses pymoo's proven HV indicator for correctness.
 98
 99    Args:
100        pareto_front: Points on the Pareto front (n_points x n_objectives)
101        reference_point: Reference point (worst possible values) for each objective
102
103    Returns:
104        Hypervolume value
105
106    Raises:
107        ImportError: If pymoo is not installed
108        ValueError: If reference point doesn't dominate all points
109    """
110    try:
111        from pymoo.indicators.hv import HV
112    except ImportError as e:
113        raise ImportError(
114            "Hypervolume calculation requires pymoo. " "Install with: pip install pymoo"
115        ) from e
116
117    # Validate reference point dominates all points
118    if not np.all(pareto_front <= reference_point):
119        raise ValueError("Reference point must dominate all points in the Pareto front")
120
121    hv_indicator = HV(ref_point=reference_point)
122    return float(hv_indicator(pareto_front))

def hedges_g( group_a: numpy.ndarray, group_b: numpy.ndarray, paired: bool = False, ci: float = 0.95, n_bootstrap: int = 10000) -> EffectSize: 
 53def hedges_g(
 54    group_a: np.ndarray,
 55    group_b: np.ndarray,
 56    paired: bool = False,
 57    ci: float = 0.95,
 58    n_bootstrap: int = 10000,
 59) -> EffectSize:
 60    """Calculate Hedges' g with bootstrap confidence interval.
 61
 62    Hedges' g is a bias-corrected version of Cohen's d for small samples.
 63    Use paired=True when comparing the same solutions across different arenas.
 64
 65    Args:
 66        group_a: Measurements for group A
 67        group_b: Measurements for group B
 68        paired: Whether samples are paired (default: False)
 69        ci: Confidence level (default: 0.95)
 70        n_bootstrap: Number of bootstrap resamples (default: 10000)
 71
 72    Returns:
 73        EffectSize with g, CI, CLES, and interpretation
 74
 75    Note:
 76        For paired data (e.g., same solutions in different arenas),
 77        use paired=True to account for within-subject correlation.
 78        This is the **correct** approach per the PRD.
 79    """
 80    group_a = np.asarray(group_a, dtype=float)
 81    group_b = np.asarray(group_b, dtype=float)
 82
 83    if paired and len(group_a) != len(group_b):
 84        raise ValueError("Paired samples must have same length")
 85
 86    n_a, n_b = len(group_a), len(group_b)
 87
 88    if paired:
 89        # Paired design: use differences
 90        differences = group_a - group_b
 91        mean_diff = np.mean(differences)
 92        sd_diff = np.std(differences, ddof=1)
 93
 94        if sd_diff == 0:
 95            # No variance in differences
 96            g = 0.0
 97        else:
 98            # Cohen's d_z for paired samples
 99            d_z = mean_diff / sd_diff
100
101            # Hedges' correction for paired data
102            correction = 1 - (3 / (4 * n_a - 5))
103            g = d_z * correction
104
105    else:
106        # Independent samples
107        pooled_sd = _pooled_std(group_a, group_b)
108
109        if pooled_sd == 0:
110            # No variance
111            g = 0.0
112        else:
113            # Cohen's d
114            d = (np.mean(group_a) - np.mean(group_b)) / pooled_sd
115
116            # Hedges' correction for small samples
117            correction = 1 - (3 / (4 * (n_a + n_b) - 9))
118            g = d * correction
119
120    # Bootstrap confidence interval
121    ci_lower, ci_upper = _bootstrap_ci(
122        group_a, group_b, paired=paired, ci=ci, n_bootstrap=n_bootstrap
123    )
124
125    # Common Language Effect Size
126    cles = _compute_cles(group_a, group_b)
127
128    # Interpretation (software engineering norms)
129    interpretation = _interpret_hedges_g(g)
130
131    return EffectSize(
132        metric="hedges_g_paired" if paired else "hedges_g",
133        value=g,
134        ci_lower=ci_lower,
135        ci_upper=ci_upper,
136        cles=cles,
137        interpretation=interpretation,
138        sample_sizes=(n_a, n_b),
139    )

Hedges' g is a bias-corrected version of Cohen's d for small samples. Use paired=True when comparing the same solutions across different arenas.

Args: group_a: Measurements for group A group_b: Measurements for group B paired: Whether samples are paired (default: False) ci: Confidence level (default: 0.95) n_bootstrap: Number of bootstrap resamples (default: 10000)
