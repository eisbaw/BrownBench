 22class DUTRegistry:
 23    """Thread-safe registry of available DUTs.
 24
 25    This registry is an optional convenience for dynamic DUT discovery.
 26    All DUT classes can be instantiated and used directly without registration.
 27
 28    Thread safety: All operations on the registry are protected by a lock
 29    to support concurrent registration and lookups.
 30    """
 31
 32    _registry: dict[str, type[CodeGenerationApproach]] = {}
 33    _lock = threading.RLock()
 34    _audit_on_register: bool = False  # Set to True to audit DUTs on registration
 35
 36    @classmethod
 37    def register(
 38        cls, dut_class: type[CodeGenerationApproach], audit: bool | None = None
 39    ) -> type[CodeGenerationApproach]:
 40        """Register a DUT implementation in a thread-safe manner.
 41
 42        Args:
 43            dut_class: DUT class to register
 44            audit: Whether to audit the DUT for security issues. If None, uses _audit_on_register.
 45
 46        Returns:
 47            The same class (for use as decorator)
 48
 49        Raises:
 50            DUTError: If DUT class doesn't define APPROACH_ID or fails security audit
 51        """
 52        # Get ID from class attribute (no instance needed)
 53        dut_id = dut_class.APPROACH_ID
 54        if not dut_id:
 55            raise DUTError(f"{dut_class.__name__} must define APPROACH_ID class attribute")
 56
 57        # Optionally audit the DUT for security issues
 58        should_audit = audit if audit is not None else cls._audit_on_register
 59        if should_audit:
 60            from codegen_bench.security.auditor import DUTAuditor
 61
 62            auditor = DUTAuditor()
 63            report = auditor.audit_dut(dut_class)
 64
 65            logger.info(
 66                "dut_audit_completed",
 67                dut_id=dut_id,
 68                passed=report.passed,
 69                num_findings=len(report.findings),
 70            )
 71
 72            if not report.passed:
 73                logger.error(
 74                    "dut_audit_failed",
 75                    dut_id=dut_id,
 76                    summary=report.get_summary(),
 77                    findings=[
 78                        f.description
 79                        for f in report.findings
 80                        if f.risk_level.value in ("critical", "high")
 81                    ],
 82                )
 83                raise DUTError(f"DUT {dut_id} failed security audit: {report.get_summary()}")
 84
 85        with cls._lock:
 86            cls._registry[dut_id] = dut_class
 87
 88        return dut_class

 35    def _read_file_safely(self, file_path: Path, max_lines: int = 200) -> str:
 36        """Read file content safely with error handling.
 37
 38        Args:
 39            file_path: Path to file to read
 40            max_lines: Maximum lines to read
 41
 42        Returns:
 43            File content or error message
 44        """
 45        try:
 46            if not file_path.exists():
 47                return f"[File not found: {file_path}]"
 48
 49            with open(file_path, encoding="utf-8", errors="ignore") as f:
 50                lines = f.readlines()[:max_lines]
 51                content = "".join(lines)
 52
 53            if len(lines) == max_lines:
 54                content += f"\n... (truncated, showing first {max_lines} lines)"
 55
 56            return content
 57        except Exception as e:
 58            return f"[Error reading {file_path}: {e}]"
 59
 60    def _get_codebase_context(self, task: Task) -> str:
 61        """Get relevant codebase context by reading key files.
 62
 63        Args:
 64            task: Task with key_files to analyze
 65
 66        Returns:
 67            Formatted codebase context or empty string
 68        """
 69        if not self.include_codebase_context or not task.key_files:
 70            return ""
 71
 72        context_parts = ["\n## CODEBASE CONTEXT\n"]
 73        context_parts.append(
 74            "Here are the current versions of the key files you'll be modifying:\n"
 75        )
 76
 77        files_to_read = task.key_files[: self.max_context_files]
 78
 79        for file_path_str in files_to_read:
 80            file_path = task.codebase_path / file_path_str
 81            content = self._read_file_safely(file_path)
 82
 83            context_parts.append(f"\n### {file_path_str}\n")
 84            context_parts.append(f"```\n{content}\n```\n")
 85
 86        if len(task.key_files) > self.max_context_files:
 87            remaining = len(task.key_files) - self.max_context_files
 88            context_parts.append(
 89                f"\n(Plus {remaining} more files not shown - focus on the above)\n"
 90            )
 91
 92        return "".join(context_parts)

 80    def solve(self, task: Task, seed: int) -> Solution:
 81        """Generate solution with simple prompt.
 82
 83        Args:
 84            task: Task to solve
 85            seed: Random seed (note: Claude API may not be fully deterministic)
 86
 87        Returns:
 88            Solution with generated code
 89        """
 90        logger.info(
 91            "starting_solution_generation",
 92            dut=self.get_metadata().id,
 93            task_id=task.id,
 94            seed=seed,
 95            model=self.model,
 96        )
 97
 98        try:
 99            start_time = time.time()
100
101            # Build simple prompt
102            prompt = self._build_prompt(task)
103            logger.debug("prompt_built", task_id=task.id, prompt_length=len(prompt))
104
105            # Single API call with retry
106            response = self._call_api_with_retry(prompt)
107
108            elapsed = time.time() - start_time
109
110            # Parse response to extract code changes - extract text from first content block
111            first_block = response.content[0]
112            if isinstance(first_block, TextBlock):
113                response_text = first_block.text
114            else:
115                raise DUTParsingError(f"Unexpected response type: {type(first_block)}")
116
117            changed_files = self._parse_code_changes(response_text)
118            logger.debug(
119                "code_extracted",
120                task_id=task.id,
121                num_files=len(changed_files),
122                files=list(changed_files.keys()),
123            )
124
125            # Calculate cost
126            cost = self._calculate_cost(response.usage.input_tokens, response.usage.output_tokens)
127
128            logger.info(
129                "solution_generated",
130                dut=self.get_metadata().id,
131                task_id=task.id,
132                seed=seed,
133                elapsed_seconds=elapsed,
134                input_tokens=response.usage.input_tokens,
135                output_tokens=response.usage.output_tokens,
136                cost_usd=cost,
137                num_files=len(changed_files),
138            )
139
140            return Solution(
141                approach_id=self.get_metadata().id,
142                task_id=task.id,
143                seed=seed,
144                timestamp=datetime.now(UTC),
145                changed_files=changed_files,
146                generation_time_seconds=elapsed,
147                total_tokens=response.usage.input_tokens + response.usage.output_tokens,
148                prompt_tokens=response.usage.input_tokens,
149                completion_tokens=response.usage.output_tokens,
150                api_calls=1,
151                cost_usd=cost,
152            )
153        except (anthropic.APIError, RetryError) as e:
154            logger.error(
155                "api_error",
156                dut=self.get_metadata().id,
157                task_id=task.id,
158                seed=seed,
159                error=str(e),
160                error_type=type(e).__name__,
161            )
162            # Extract original error from RetryError if applicable
163            if isinstance(e, RetryError):
164                raise DUTAPIError(f"API call failed for task {task.id} after retries: {e}") from e
165            else:
166                raise DUTAPIError(f"API call failed for task {task.id}: {e}") from e
167        except (KeyError, AttributeError, ValueError) as e:
168            logger.error(
169                "response_parsing_failed",
170                dut=self.get_metadata().id,
171                task_id=task.id,
172                seed=seed,
173                error=str(e),
174                error_type=type(e).__name__,
175            )
176            raise DUTParsingError(f"Failed to parse API response for task {task.id}: {e}") from e

178    def _call_api_with_retry(self, prompt: str) -> anthropic.types.Message:
179        """Call Anthropic API with retry logic and rate limiting.
180
181        Rate limiting is applied once before retries begin. If the API call fails
182        after all retries, the rate limiter slot is released for idempotency.
183
184        Args:
185            prompt: The prompt to send
186
187        Returns:
188            API response
189
190        Raises:
191            anthropic.APIError: If all retry attempts fail
192        """
193        # Apply rate limiting before any retry attempts
194        self.rate_limiter.acquire()
195
196        @retry(stop=stop_after_attempt(3), wait=wait_exponential(min=1, max=10))  # type: ignore[misc]
197        def _call_api() -> anthropic.types.Message:
198            """Inner function with retry logic that doesn't re-acquire rate limit."""
199            return self.client.messages.create(
200                model=self.model,
201                max_tokens=self.max_tokens,
202                temperature=self.temperature,
203                messages=[{"role": "user", "content": prompt}],
204            )
205
206        try:
207            return _call_api()
208        except Exception:
209            # Release rate limiter slot on failure for idempotency
210            self.rate_limiter.release()
211            raise

214        """Build minimal prompt for task.

247}}

252        """Extract file changes from response.

262        return CodeExtractor.extract_all(response)

264    def _calculate_cost(self, input_tokens: int, output_tokens: int) -> float:
265        """Calculate API cost based on token usage.
266
267        Uses pricing as of January 2025:
268        - Claude Sonnet 4.5: $3/M input, $15/M output
269
270        Args:
271            input_tokens: Number of input tokens
272            output_tokens: Number of output tokens
273
274        Returns:
275            Cost in USD
276        """
277        input_cost = (input_tokens / 1000) * CLAUDE_SONNET_4_5_INPUT_COST_PER_1K_TOKENS
278        output_cost = (output_tokens / 1000) * CLAUDE_SONNET_4_5_OUTPUT_COST_PER_1K_TOKENS
279
280        return input_cost + output_cost
