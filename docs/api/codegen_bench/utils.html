 15    """Token bucket rate limiter for API calls.
 16
 17    Thread-safe rate limiter implementing the token bucket algorithm.
 18    Supports both requests-per-minute and tokens-per-minute limits.
 19
 20    Example:
 21        >>> limiter = RateLimiter(max_requests_per_minute=50)
 22        >>> limiter.acquire()  # Blocks if rate limit exceeded
 23        >>> # Make API call
 24    """

 26    def __init__(
 27        self,
 28        max_requests_per_minute: int = 50,
 29        max_tokens_per_minute: int | None = None,
 30    ):
 31        """Initialize rate limiter.
 32
 33        Args:
 34            max_requests_per_minute: Maximum requests allowed per minute (must be > 0)
 35            max_tokens_per_minute: Maximum tokens allowed per minute (must be > 0 if set)
 36
 37        Raises:
 38            ValueError: If rate limits are invalid (zero or negative)
 39        """
 40        if max_requests_per_minute <= 0:
 41            raise ValueError(
 42                f"max_requests_per_minute must be positive, got {max_requests_per_minute}"
 43            )
 44
 45        if max_tokens_per_minute is not None and max_tokens_per_minute <= 0:
 46            raise ValueError(f"max_tokens_per_minute must be positive, got {max_tokens_per_minute}")
 47
 48        self.max_requests_per_minute = max_requests_per_minute
 49        self.max_tokens_per_minute = max_tokens_per_minute
 50
 51        # Request rate limiting
 52        self._request_times: deque[float] = deque()
 53
 54        # Token rate limiting (if enabled)
 55        self._token_times: deque[tuple[float, int]] = deque()
 56
 57        # Single lock for atomic acquire() operations
 58        self._lock = threading.Lock()
 59
 60    def acquire(self, tokens: int = 0) -> None:
 61        """Acquire permission to make an API call.
 62
 63        Blocks until rate limit allows the request.
 64        Uses a single lock to ensure atomic operations and timestamp consistency.
 65
 66        Args:
 67            tokens: Number of tokens this request will use (for token-based limiting)
 68
 69        Raises:
 70            ValueError: If tokens is negative
 71        """
 72        if tokens < 0:
 73            raise ValueError(f"Token count must be non-negative, got {tokens}")
 74
 75        with self._lock:
 76            # Use single timestamp for consistency across both limits
 77            now = time.time()
 78            cutoff = now - 60.0  # 1 minute ago
 79            max_sleep_time = 0.0
 80
 81            # Check request-based rate limiting
 82            # Remove old timestamps
 83            while self._request_times and self._request_times[0] < cutoff:
 84                self._request_times.popleft()
 85
 86            # Check if we need to wait for request limit
 87            if len(self._request_times) >= self.max_requests_per_minute:
 88                # Wait until oldest request is outside the window
 89                request_sleep = self._request_times[0] + 60.0 - now
 90                max_sleep_time = max(max_sleep_time, request_sleep)
 91
 92            # Check token-based rate limiting (if enabled)
 93            if self.max_tokens_per_minute is not None and tokens > 0:
 94                # Remove old token records
 95                while self._token_times and self._token_times[0][0] < cutoff:
 96                    self._token_times.popleft()
 97
 98                # Calculate current token usage
 99                current_tokens = sum(t[1] for t in self._token_times)
100
101                # Check if we need to wait for token limit
102                if current_tokens + tokens > self.max_tokens_per_minute:
103                    # Find when we'll have enough tokens
104                    needed = current_tokens + tokens - self.max_tokens_per_minute
105                    freed = 0
106
107                    for timestamp, token_count in self._token_times:
108                        freed += token_count
109                        if freed >= needed:
110                            token_sleep = timestamp + 60.0 - now
111                            max_sleep_time = max(max_sleep_time, token_sleep)
112                            break
113
114            # Sleep outside the lock if needed, then re-acquire and record
115            if max_sleep_time > 0:
116                logger.info(
117                    "rate_limit_hit",
118                    sleep_seconds=round(max_sleep_time, 2),
119                    requests_in_window=len(self._request_times),
120                    max_requests_per_minute=self.max_requests_per_minute,
121                    tokens_requested=tokens if self.max_tokens_per_minute else None,
122                    tokens_in_window=(
123                        sum(t[1] for t in self._token_times) if self.max_tokens_per_minute else None
124                    ),
125                    max_tokens_per_minute=self.max_tokens_per_minute,
126                )
127                # Release lock before sleeping to allow other threads to proceed
128                self._lock.release()
129                try:
130                    time.sleep(max_sleep_time)
131                finally:
132                    self._lock.acquire()
133
134                # Re-compute timestamp and clean up after sleeping
135                now = time.time()
136                cutoff = now - 60.0
137
138                # Clean up old request timestamps
139                while self._request_times and self._request_times[0] < cutoff:
140                    self._request_times.popleft()
141
142                # Clean up old token records
143                if self.max_tokens_per_minute is not None:
144                    while self._token_times and self._token_times[0][0] < cutoff:
145                        self._token_times.popleft()
146
147                # Re-verify limits after sleep (race condition: another thread may have consumed slots)
148                # If still over limit, wait again
149                while True:
150                    # ALWAYS update timestamp and cleanup at start of each iteration
151                    now = time.time()
152                    cutoff = now - 60.0
153
154                    # Clean up ALL old records
155                    while self._request_times and self._request_times[0] < cutoff:
156                        self._request_times.popleft()
157
158                    if self.max_tokens_per_minute is not None:
159                        while self._token_times and self._token_times[0][0] < cutoff:
160                            self._token_times.popleft()
161
162                    # Check request limit
163                    if len(self._request_times) >= self.max_requests_per_minute:
164                        wait_time = self._request_times[0] + 60.0 - now
165                        if wait_time > 0:
166                            logger.debug("rate_limit_race_detected", reason="request_limit")
167                            self._lock.release()
168                            try:
169                                time.sleep(wait_time)
170                            finally:
171                                self._lock.acquire()
172                            continue  # Re-check from top with fresh timestamp
173
174                    # Check token limit
175                    if self.max_tokens_per_minute is not None and tokens > 0:
176                        current_tokens = sum(t[1] for t in self._token_times)
177                        if current_tokens + tokens > self.max_tokens_per_minute:
178                            # Find when we'll have enough tokens
179                            needed = current_tokens + tokens - self.max_tokens_per_minute
180                            freed = 0
181                            wait_time = 0.0
182
183                            for timestamp, token_count in self._token_times:
184                                freed += token_count
185                                if freed >= needed:
186                                    wait_time = timestamp + 60.0 - now
187                                    break
188
189                            if wait_time > 0:
190                                logger.debug("rate_limit_race_detected", reason="token_limit")
191                                self._lock.release()
192                                try:
193                                    time.sleep(wait_time)
194                                finally:
195                                    self._lock.acquire()
196                                continue  # Re-check from top with fresh timestamp
197
198                    # Both limits satisfied, break out
199                    break
200
201            # Record this request with consistent timestamp
202            self._request_times.append(now)
203            if self.max_tokens_per_minute is not None and tokens > 0:
204                self._token_times.append((now, tokens))
205
206    def release(self, tokens: int = 0) -> None:
207        """Release a previously acquired request slot.
208
209        Call this when an API request fails and you want to "return" the
210        rate limit slot for retry idempotency. This removes the most recent
211        request from tracking.
212
213        Args:
214            tokens: Number of tokens to release (must match acquire call)
215
216        Raises:
217            ValueError: If tokens is negative, no requests to release,
218                       or token mismatch
219        """
220        if tokens < 0:
221            raise ValueError(f"Token count must be non-negative, got {tokens}")
222
223        with self._lock:
224            if not self._request_times:
225                raise ValueError("No requests to release")
226
227            # Remove the most recent request
228            self._request_times.pop()
229
230            # Remove the most recent token usage if applicable
231            # Only validate/release tokens if token limiting is enabled AND tokens were used
232            if tokens > 0:
233                if self.max_tokens_per_minute is None:
234                    # Token limiting not enabled but caller tried to release tokens
235                    raise ValueError(
236                        f"Cannot release {tokens} tokens: token limiting is not enabled"
237                    )
238
239                if not self._token_times:
240                    raise ValueError("No token requests to release")
241
242                # Verify we're releasing the right amount
243                _, recorded_tokens = self._token_times[-1]
244                if recorded_tokens != tokens:
245                    raise ValueError(
246                        f"Token mismatch: tried to release {tokens} tokens, "
247                        f"but last request used {recorded_tokens} tokens"
248                    )
249
250                self._token_times.pop()
251
252    def reset(self) -> None:
253        """Reset all rate limit counters.
254
255        Useful for testing or manual reset scenarios.
256        """
257        with self._lock:
258            self._request_times.clear()
259            self._token_times.clear()
260
261    def get_stats(self) -> dict[str, Any]:
262        """Get current rate limiter statistics.
263
264        Returns:
265            Dictionary with current request and token counts
266        """
267        with self._lock:
268            now = time.time()
269            cutoff = now - 60.0
270            recent_requests = sum(1 for t in self._request_times if t >= cutoff)
271
272            stats: dict[str, Any] = {
273                "requests_in_last_minute": recent_requests,
274                "max_requests_per_minute": self.max_requests_per_minute,
275            }
276
277            if self.max_tokens_per_minute is not None:
278                recent_tokens = sum(t[1] for t in self._token_times if t[0] >= cutoff)
279                stats["tokens_in_last_minute"] = recent_tokens
280                stats["max_tokens_per_minute"] = self.max_tokens_per_minute
281
282        return stats

 72class ShutdownHandler:
 73    """Handle shutdown signals gracefully.
 74
 75    Registers signal handlers for SIGINT and SIGTERM that set a thread-safe
 76    shutdown flag. Long-running operations can check this flag and perform
 77    cleanup before exiting.
 78
 79    The signal handler performs ONLY atomic operations (setting threading.Event).
 80    All I/O, logging, and cleanup must happen in application code, not in
 81    signal handlers.
 82    """
 83
 84    def __init__(self) -> None:
 85        """Initialize shutdown handler.
 86
 87        Creates an Event for thread-safe shutdown signaling.
 88        Does not register signal handlers yet - call register() or use
 89        as context manager.
 90        """
 91        self._shutdown_event = threading.Event()
 92        self._original_handlers: dict[int, signal.Handlers] = {}
 93        self._logged_shutdown = False  # Track if we've logged shutdown
 94
 95    @property
 96    def shutdown_requested(self) -> bool:
 97        """Check if shutdown has been requested.
 98
 99        Thread-safe: can be called from any thread.
100        Safe to call frequently in tight loops.
101
102        Returns:
103            True if SIGINT or SIGTERM was received, False otherwise
104        """
105        is_shutdown = self._shutdown_event.is_set()
106
107        # Log once when shutdown first detected (not in signal handler!)
108        if is_shutdown and not self._logged_shutdown:
109            self._logged_shutdown = True
110            logger.info("Shutdown requested via signal, initiating graceful shutdown")
111
112        return is_shutdown
113
114    @property
115    def is_registered(self) -> bool:
116        """Check if signal handlers are currently registered.
117
118        Returns:
119            True if signal handlers are active, False otherwise
120        """
121        return bool(self._original_handlers)
122
123    def register(self) -> None:
124        """Register signal handlers.
125
126        Registers handlers for SIGINT (Ctrl+C) and SIGTERM.
127        On Windows, SIGTERM may not be available.
128
129        Raises:
130            RuntimeError: If handlers are already registered
131        """
132        if self.is_registered:
133            raise RuntimeError(
134                "Shutdown handlers already registered. "
135                "Only one handler should be active at a time."
136            )
137
138        # Register handlers and save originals
139        self._original_handlers[signal.SIGINT] = signal.signal(signal.SIGINT, self._handle_signal)
140
141        # SIGTERM might not be available on Windows
142        if hasattr(signal, "SIGTERM"):
143            self._original_handlers[signal.SIGTERM] = signal.signal(
144                signal.SIGTERM, self._handle_signal
145            )
146
147        logger.debug("Shutdown handlers registered for SIGINT and SIGTERM")
148
149    def unregister(self) -> None:
150        """Restore original signal handlers.
151
152        Safe to call even if handlers were never registered.
153        Idempotent: safe to call multiple times.
154        """
155        if not self.is_registered:
156            return
157
158        # Restore all original handlers
159        for sig, handler in self._original_handlers.items():
160            signal.signal(sig, handler)
161
162        self._original_handlers.clear()
163        logger.debug("Shutdown handlers unregistered")
164
165    def reset(self) -> None:
166        """Reset shutdown flag.
167
168        Clears the shutdown event, allowing the handler to be reused.
169        Useful for testing or reusing the handler after a shutdown.
170
171        Note: Does not unregister signal handlers.
172        """
173        self._shutdown_event.clear()
174        self._logged_shutdown = False
175        logger.debug("Shutdown flag reset")
176
177    def _handle_signal(self, signum: int, frame: Any) -> None:
178        """Handle shutdown signal.
179
180        CRITICAL: This runs in signal handler context with severe restrictions.
181        - NO I/O (no print, no logging, no file operations)
182        - NO locks (no threading.Lock, no queue operations)
183        - NO memory allocation (keep it simple)
