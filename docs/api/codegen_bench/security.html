349    def _forward_request(
350        self,
351        method: str,
352        path: str,
353        headers: dict[str, str],
354        body: bytes,
355        dut_id: str,
356    ) -> httpx.Response:
357        """Forward request to upstream API.
358
359        Args:
360            method: HTTP method
361            path: Request path
362            headers: Request headers
363            body: Request body
364            dut_id: DUT identifier for budget tracking
365
366        Returns:
367            Response from upstream API
368
369        Raises:
370            APIProxyError: If forwarding fails
371        """
372        # Build upstream URL
373        url = f"{self._config.api_base_url}{path}"
374
375        # Replace temporary key with real key
376        headers = dict(headers)  # Copy to avoid mutating caller's dict
377        headers["x-api-key"] = self._config.real_api_key
378
379        # Remove hop-by-hop headers
380        for hop_header in ("host", "connection", "keep-alive", "transfer-encoding"):
381            headers.pop(hop_header, None)
382
383        logger.debug(
384            "forwarding_request",
385            dut_id=dut_id,
386            method=method,
387            url=url,
388        )
389
390        # Forward request
391        response = self._http_client.request(
392            method=method,
393            url=url,
394            headers=headers,
395            content=body,
396        )
397
398        # Track cost if budget enforcer is configured
399        if self._config.budget_enforcer:
400            cost = self._estimate_cost_from_response(response)
401            if cost > 0:
402                try:
403                    self._config.budget_enforcer.record_cost(dut_id, cost)
404                    logger.info(
405                        "cost_tracked",
406                        dut_id=dut_id,
407                        cost_usd=cost,
408                    )
409                except Exception as e:
410                    logger.error(
411                        "cost_tracking_failed",
412                        dut_id=dut_id,
413                        error=str(e),
414                    )
415                    # Re-raise to fail fast on budget violations
416                    raise
417
418        return response

420    def _estimate_cost_from_response(self, response: httpx.Response) -> float:
421        """Estimate cost from API response.
422
423        This is a simple estimation based on response headers or body.
424        For Anthropic API, we parse the usage field from the JSON response.
425
426        Args:
427            response: API response
428
429        Returns:
430            Estimated cost in USD
431        """
432        try:
433            # For Anthropic API, parse usage from JSON response
434            if response.status_code == 200:
435                data = response.json()
436                usage = data.get("usage", {})
437                input_tokens = usage.get("input_tokens", 0)
438                output_tokens = usage.get("output_tokens", 0)
439
440                # Use Claude Sonnet 4.5 pricing (matching SimpleChatDUT)
441                # $3/M input, $15/M output
442                input_cost = (input_tokens / 1000) * 0.003
443                output_cost = (output_tokens / 1000) * 0.015
444
445                return input_cost + output_cost
446
447        except Exception as e:
448            logger.warning(
449                "cost_estimation_failed",
450                error=str(e),
451                note="Cost tracking may be inaccurate",
452            )
453
454        return 0.0
