350    def _compute_arena_breakdowns(self, result: TournamentResult) -> list[dict[str, Any]]:
351        """Compute per-arena statistics.
352
353        Args:
354            result: Tournament results
355
356        Returns:
357            List of arena breakdown dictionaries
358        """
359        breakdowns = []
360
361        for arena in result.arenas:
362            arena_comparisons = [c for c in result.comparisons if c[2] == arena]
363
364            if not arena_comparisons:
365                continue
366
367            # Compute agreement rate
368            agreement_rates = [judgment.agreement_rate for _, _, _, judgment in arena_comparisons]
369            avg_agreement = sum(agreement_rates) / len(agreement_rates)
370
371            # Count ties
372            ties = sum(1 for _, _, _, judgment in arena_comparisons if judgment.winner == "TIE")
373
374            breakdowns.append(
375                {
376                    "name": arena,
377                    "total_comparisons": len(arena_comparisons),
378                    "avg_agreement": avg_agreement,
379                    "tie_count": ties,
380                    "tie_rate": ties / len(arena_comparisons) if arena_comparisons else 0.0,
381                }
382            )
383
384        return breakdowns
